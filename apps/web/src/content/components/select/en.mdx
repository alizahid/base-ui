export const items = { apple: 'Apple', banana: 'Banana', cherry: 'Cherry', carrot: 'Carrot', potato: 'Potato' }

export const currencies = [
  { code: 'usd', flag: 'us', name: 'US Dollar' },
  { code: 'eur', flag: 'eu', name: 'Euro' },
  { code: 'aed', flag: 'ae', name: 'UAE Dirham' },
  { code: 'gbp', flag: 'gb', name: 'British Pound' },
]

export const Flag = ({ flag }) => (
  <img alt={flag} className="size-4 rounded-full" src={`https://flags.willa.app/flags/${flag}.svg`} />
)

{/* what the trigger shows for each value: the flag and the code */}
export const currencyItems = Object.fromEntries(
  currencies.map((currency) => [
    currency.code,
    <span className="flex items-center gap-2" key={currency.code}>
      <Flag flag={currency.flag} />
      <span className="uppercase">{currency.code}</span>
    </span>,
  ]),
)
