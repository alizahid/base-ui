import { useText } from '@/components/text'
import { title } from '@/intl/metadata'

export const generateMetadata = title('select')

export const useItems = () => ({ apple: useText('common.apple'), banana: useText('common.banana'), cherry: useText('common.cherry'), carrot: useText('common.carrot'), potato: useText('select.potato') })

export const useCurrencies = () => [
  { code: 'usd', flag: 'us', name: useText('select.us-dollar') },
  { code: 'eur', flag: 'eu', name: useText('select.euro') },
  { code: 'aed', flag: 'ae', name: useText('select.uae-dirham') },
  { code: 'gbp', flag: 'gb', name: useText('select.british-pound') },
]

export const Flag = ({ flag }) => (
  <img alt={flag} className="size-4 rounded-full" src={`https://flags.willa.app/flags/${flag}.svg`} />
)

{/* what the trigger shows for each value: the flag and the code */}
export const useCurrencyItems = () => Object.fromEntries(
  useCurrencies().map((currency) => [
    currency.code,
    <span className="flex items-center gap-2" key={currency.code}>
      <Flag flag={currency.flag} />
      <span className="uppercase">{currency.code}</span>
    </span>,
  ]),
)
